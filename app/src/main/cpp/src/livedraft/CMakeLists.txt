set(CRISPASR_SO "${CMAKE_SOURCE_DIR}/../../jniLibs/arm64-v8a/libcrispasr.so")
add_library(livedraft_cosy SHARED livedraft_cosy.cpp)
target_compile_features(livedraft_cosy PRIVATE cxx_std_17)
if(EXISTS "${CRISPASR_SO}")
    target_compile_definitions(livedraft_cosy PRIVATE LIVE_DRAFT_COSY=1)
    add_library(crispasr_prebuilt SHARED IMPORTED)
    set_target_properties(crispasr_prebuilt PROPERTIES IMPORTED_LOCATION "${CRISPASR_SO}")
    target_link_libraries(livedraft_cosy PRIVATE crispasr_prebuilt log)
else()
    target_link_libraries(livedraft_cosy PRIVATE log)
endif()
